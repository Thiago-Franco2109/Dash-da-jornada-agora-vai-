pagamento
 
online
 
 
  
 
CREATE  OR  REPLACE  VIEW  Comissao_View  AS  
SELECT  
    p.id  AS  pedido_id,  
    p.estabelecimento_id  AS  estabelecimento_id,  
    e.nome  AS  estabelecimento,  
    l.id  AS  cidade_id,  
    l.nome  AS  cidade,  
    p.data  AS  data,  
    p.status  AS  status,  
 
    --  Cálculo  da  Comissão  Bruta  para  o  pedido  
        IF(p.status  IN  (1,  2),  
            IF(p.data  >=  '2024-04-01  00:00:00',  
                IF(p.comissao_entrega  =  1,  
                    (p.subtotal  -  IF(p.estabelecimento_cupom  =  1,  p.desconto,  0)  +  p.taxa_entrega  
+
 
p.reembolso_promocao)
 
*
 
(p.comissao
 
/
 
100),
 
                    (p.subtotal  -  IF(p.estabelecimento_cupom  =  1,  p.desconto,  0)  +  
p.reembolso_promocao)
 
*
 
(p.comissao
 
/
 
100)
 
                ),  
                IF(p.comissao_entrega  =  1,  
                    (p.subtotal  -  IF(p.estabelecimento_cupom  =  1,  p.desconto,  0)  +  p.taxa_entrega)  
*
 
(p.comissao
 
/
 
100),
 
                    (p.subtotal  -  IF(p.estabelecimento_cupom  =  1,  p.desconto,  0))  *  (p.comissao  /  
100)
 
                )  
            ),  
            0  
        )  AS  comissao_bruta,  
 
    --  Cálculo  da  Comissão  Líquida  para  o  pedido  
  
 
        IF(p.status  IN  (1,  2),  
            IF(p.data  >=  '2024-04-01  00:00:00',  
                IF(p.comissao_entrega  =  1,  
                    ((p.subtotal  -  IF(p.estabelecimento_cupom  =  1,  p.desconto,  0)  +  p.taxa_entrega  
+
 
p.reembolso_promocao)
 
*
 
(p.comissao
 
/
 
100))
 
                    -  (IF(p.estabelecimento_cupom  =  0,  p.desconto,  0)  +  p.reembolso_promocao),  
                    ((p.subtotal  -  IF(p.estabelecimento_cupom  =  1,  p.desconto,  0)  +  
p.reembolso_promocao)
 
*
 
(p.comissao
 
/
 
100))
 
                    -  (IF(p.estabelecimento_cupom  =  0,  p.desconto,  0)  +  p.reembolso_promocao)  
                ),  
                IF(p.comissao_entrega  =  1,  
                    ((p.subtotal  -  IF(p.estabelecimento_cupom  =  1,  p.desconto,  0)  +  
p.taxa_entrega)
 
*
 
(p.comissao
 
/
 
100))
 
                    -  IF(p.estabelecimento_cupom  =  0,  p.desconto,  0),  
                    ((p.subtotal  -  IF(p.estabelecimento_cupom  =  1,  p.desconto,  0))  *  (p.comissao  /  
100))
 
                    -  IF(p.estabelecimento_cupom  =  0,  p.desconto,  0)  
                )  
            ),  
            0  
        )  AS  comissao_liquida,  
 
 --  Cálculo  do  GMV  Bruto  para  o  pedido  
        IF(p.status  IN  (1,  2),  
            IF(p.data  >=  '2024-04-01  00:00:00',  
                (p.subtotal  -  IF(p.estabelecimento_cupom  =  1,  p.desconto,  0)  +  p.taxa_entrega  +  
p.reembolso_promocao),
 
                (p.subtotal  -  IF(p.estabelecimento_cupom  =  1,  p.desconto,  0)  +  p.taxa_entrega)  
            ),  
            0  
  
 
        )  AS  gmv_bruto,  
 
    --  Cálculo  da  GMV  Líquido  para  o  pedido  
        IF(p.status  IN  (1,  2),  
           p.total,  
           --  (p.subtotal  +  p.taxa_entrega  -  p.desconto),  
            0  
        )  AS  gmv_liquido,  
 
        IF(p.status  IN  (1,  2),  IF(p.estabelecimento_cupom  =  1,  p.desconto,  0),0)  AS  
cupom_parceiro,
 
        IF(p.status  IN  (1,  2),  IF(p.estabelecimento_cupom  =  0,  p.desconto,  0),0)  AS  
cupom_bigou,
 
        IF(p.status  IN  (1,  2),  p.reembolso_promocao,  0)  AS  reembolso,  
  IF(p.status  IN  (1,  2),  p.tipo_pgt_online,  0)  AS  tipo_pgt,  
 
 --  Cálculo  da  Taxa  de  Pagamento  Online  
        IF(p.status  IN  (1,  2),  
            IF(p.data  >=  '2024-04-01  00:00:00',  
                IF(p.comissao_entrega  =  1,  
                    (p.subtotal  -  IF(p.estabelecimento_cupom  =  1,  p.desconto,  0)  +  p.taxa_entrega  
+
 
p.reembolso_promocao)
 
*
 
(p.taxa_pgt_online
 
/
 
100),
 
                    (p.subtotal  -  IF(p.estabelecimento_cupom  =  1,  p.desconto,  0)  +  
p.reembolso_promocao)
 
*
 
(p.taxa_pgt_online
 
/
 
100)
 
                ),  
                IF(p.comissao_entrega  =  1,  
                    (p.subtotal  -  IF(p.estabelecimento_cupom  =  1,  p.desconto,  0)  +  p.taxa_entrega)  
*
 
(p.taxa_pgt_online
 
/
 
100),
 
                    (p.subtotal  -  IF(p.estabelecimento_cupom  =  1,  p.desconto,  0))  *  
(p.taxa_pgt_online
 
/
 
100)
 
                )  
  
 
            ),  
            0  
        )  AS  taxa_pgt_online,  
 
 CASE  
        WHEN  p.status  IN  (1,  2)  THEN  
            CASE  
                WHEN  p.tipo_pgt_online  LIKE  'offline'  THEN  0  
                WHEN  p.tipo_pgt_online  LIKE  'pix-sicoob'  THEN  0  
                WHEN  p.tipo_pgt_online  LIKE  'pix-pagbank'  THEN  0.10  
                WHEN  p.tipo_pgt_online  LIKE  'pix-mercadopago'  THEN  p.total  *  0.0062  
                WHEN  p.tipo_pgt_online  LIKE  'picpay'  THEN  p.total  *  0.0219  
                WHEN  p.tipo_pgt_online  LIKE  'nupay'  THEN  p.total  *  0.014  
                ELSE  0  
            END  
        ELSE  0  --  Se  p.status  não  estiver  em  (1,  2),  a  taxa  será  0.  
    END  AS  taxa_gateway_aprox  
 
 
FROM  pedido  AS  p  
LEFT  JOIN  estabelecimento  AS  e  ON  e.id  =  p.estabelecimento_id  
LEFT  JOIN  localidade  AS  l  ON  l.id  =  e.localidade_id;  
 
 
Regras  para  acessar  o  contrato  de  um  
estabelecimento
 
e
 
vice-versa
 
-  A  tabela  de  contratos  chama-se  `venda`  (alias  `v`).  
-  A  tabela  que  associa  contratos  a  estabelecimentos  é  `venda_estabelecimento`  (alias  `ve`).  
  
 
-  A  tabela  de  estabelecimentos  é  `estabelecimento`  (alias  `e`).  
 
Sempre  que  precisar  obter  dados  de  um  contrato  vinculado  a  um  estabelecimento,  gere  
uma
 
query
 
no
 
formato:
 
 
FROM  venda  AS  v  
LEFT  JOIN  venda_estabelecimento  AS  ve  
  ON  ve.venda_id  =  v.id  
LEFT  JOIN  estabelecimento  AS  e  
  ON  e.id  =  ve.estabelecimento_id  
 
E  inclua  um  `WHERE  e.id  =  ?`  para  filtrar  pelo  estabelecimento  desejado.
